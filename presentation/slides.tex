% 5. Pegasos SVM
\begin{frame}{Pegasos SVM from scratch}
  \presenter{Sahib Aliyev}
  \begin{itemize}\gap
    \item Objective, labels $\pm 1$: % src: svm.py:PegasosSVM._record_objective
      \[J(\mathbf{w},b)=\frac{\lambda}{2}\lVert\mathbf{w}\rVert^{2}
        +\frac{1}{n}\textstyle\sum_i\max\!\big(0,\,1-y_i(\mathbf{w}^{\top}\mathbf{x}_i+b)\big)\]
    \item One random example per step, $\eta_t = 1/(\lambda t)$, then % src: svm.py:PegasosSVM._fit_features
      project $\mathbf{w}$ onto $\lVert\mathbf{w}\rVert \le 1/\sqrt{\lambda}$: % src: svm.py:PegasosSVM._fit_features (radius = 1/sqrt(lambda_))
      \[\mathbf{w} \leftarrow (1 - \eta_t\lambda)\,\mathbf{w}
        + \eta_t\,\mathbf{1}\!\left[y_i(\mathbf{w}^\top\mathbf{x}_i + b) < 1\right] y_i\mathbf{x}_i\]
    \item Margin $2/\lVert\mathbf{w}\rVert$; support vectors: rows with
      $y_i(\mathbf{w}^\top\mathbf{x}_i + b) \le 1$ % src: svm.py:PegasosSVM._fit_features (support_)
    \item RFF, $D = 500$: % src: results/report_facts.csv [svm_rff_n_components]
      256~MB instead of a 32.7~GB kernel matrix % src: results/report_facts.csv [rff_matrix_mb 255.7, kernel_matrix_gb 32.69]
  \end{itemize}
\end{frame}
